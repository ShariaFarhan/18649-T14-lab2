cmake_minimum_required(VERSION 3.20.0)
set(NODE_ID 2 CACHE STRING "2=Front, 3=Rear")
if(NODE_ID STREQUAL "3")
  list(APPEND EXTRA_DTC_OVERLAY_FILE "${CMAKE_CURRENT_SOURCE_DIR}/boards/rear.overlay")
  list(APPEND EXTRA_CONF_FILE "${CMAKE_CURRENT_SOURCE_DIR}/rear.conf")
else()
  list(APPEND EXTRA_DTC_OVERLAY_FILE "${CMAKE_CURRENT_SOURCE_DIR}/boards/front.overlay")
  list(APPEND EXTRA_CONF_FILE "${CMAKE_CURRENT_SOURCE_DIR}/front.conf")
endif()
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(lab3_heartbeat)
set(NODE_ID 2 CACHE STRING "2=Front, 3=Rear")
if(NOT NODE_ID MATCHES "^[23]$")
  message(FATAL_ERROR "NODE_ID must be 2 or 3")
endif()
target_sources(app PRIVATE src/main.c src/runtime.c ../logic/blinkers/blinkers.c ../logic/system/system_state.c)
set(LAB3_PROFILE_TASK 0 CACHE STRING "Optional task execution marker (0=off,1..8)")
target_compile_definitions(app PRIVATE LAB3_PROFILE_TASK=${LAB3_PROFILE_TASK})
target_include_directories(app PRIVATE ../../pi ../../shared ../logic/blinkers ../logic/system)
target_compile_definitions(app PRIVATE NODE_ID=${NODE_ID})
if(NODE_ID STREQUAL "2")
  target_sources(app PRIVATE src/steer_task.c src/relay_task.c src/force_task.c ../logic/steering/servo.c)
  target_include_directories(app PRIVATE ../logic/steering)
else()
  target_sources(app PRIVATE src/drive_task.c ../logic/drive/motor_control.c
                             ../logic/drive/dual_speed_pid.c ../logic/instrumentation/test_points.c)
  target_include_directories(app PRIVATE ../logic/drive ../logic/instrumentation)
endif()
